ASTNode
│
├── Expression                                             <- can be evaluated as a value
│   ├── Identifier
│   ├── Literal
│   │   ├── IntLiteral
│   │   ├── FloatLiteral
│   │   ├── BoolLiteral
│   │   ├── CharLiteral
│   │   └── StringLiteral
│   │ 
│   ├── BinaryExpression
│   ├── CallExpression
│   ├── MemberExpression
│   ├── IndexExpression
│   ├── ArrayLiteral
│   └── StructLiteral
│
├── Statement                                               <- can be executed as part of a sequence of operations
│   ├── ExpressionStatement                                 <- reference to an expression
│   ├── DeclarationStatement
│   ├── AssignmentStatement
│   ├── ReturnStatement
│   └── Block
│
├── Declaration                                             <- introduces/defines a name or type
│   ├── VariableDeclaration
│   ├── FunctionDeclaration
│   ├── StructDeclaration
│   ├── EnumDeclaration
│   ├── CellDeclaration
│   └── ImplDeclaration

└── TypeNode                                                <- type syntax written in source

    ├── PrimitiveType

    │   ├── BoolType
    │   ├── CharType
    │   └── StringType
    │
    ├── ArrayType

    │
    └── NamedType                                           <- reference to a declared type
